\section{Proof of BP3: $\frac{\partial C}{\partial b_j^l} = \delta_j^l$}
We need to prove:
\begin{equation}
    \frac{\partial C}{\partial b_j^l} = \delta_j^l
\end{equation}

\subsection{Step 1: Understanding the Terms}
The weighted input to neuron $j$ in layer $l$ in $z_j^l$ is:
\begin{equation}
    z_j^l = \sum_k w_{jk}^{l} a_k^{l-1} + b_j^l
\end{equation}
where:
\begin{itemize}
    \item $w_{jk}^{l}$ is the weight from neuron $k$ in layer $l-1$ to neuron $j$ in layer $l$,
    \item $a_k^{l-1}$ is the activation of neuron $k$ in layer $l-1$,
    \item $b_j^l$ is the bias of neuron $j$ in layer $l$.
\end{itemize}
$C$ is the loss function, $\frac{\partial C}{\partial b_j^l}$ is the partial derivative of the loss function with respect to the bias. This tells us how much a change in the bias will affect the error. Finally, we have $\delta_j^l$, which is the error.

\subsection{Step 2: Applying the Chain Rule}
We want to compute:
\begin{equation}
    \frac{\partial C}{\partial b_j^l}
\end{equation}
So using the chain rule:
\begin{equation}
    \frac{\partial C}{\partial b_j^l} = \frac{\partial C}{\partial z_j^l} \cdot \frac{\partial z_j^l}{\partial b_j^l}
\end{equation}

\subsection{Step 3: Compute $\frac{\partial z_j^l}{\partial b_j^l}$}
From the equation of $z_j^l$:
\begin{equation}
    z_j^l = \sum_k w_{jk}^{l} a_k^{l-1} + b_j^l
\end{equation}
Differentiating with respect to $b_j^l$:
\begin{equation}
    \frac{\partial z_j^l}{\partial b_j^l} = 1
\end{equation}

\subsection{Step 4: Using the Definition of $\delta_j^l$}
By the backpropagation equations:
\begin{equation}
    \delta_j^l = \frac{\partial C}{\partial z_j^l}
\end{equation}
Thus, substituting into our chain rule equation:
\begin{equation}
    \frac{\partial C}{\partial b_j^l} = \delta_j^l \cdot 1 = \delta_j^l
\end{equation}
This confirms that:
\begin{equation}
    \frac{\partial C}{\partial b_j^l} = \delta_j^l
\end{equation}

\section{Proof of BP4: $\frac{\partial C}{\partial w_{jk}^l} = a_k^{l-1} \delta_j^l$}
We need to prove:
\begin{equation}
    \frac{\partial C}{\partial w_{jk}^l} = a_k^{l-1} \delta_j^l
\end{equation}

\subsection{Step 1: Understanding $z_j^l$}
The weighted input $z_j^l$ to neuron $j$ in layer $l$ is:
\begin{equation}
    z_j^l = \sum_k w_{jk}^{l} a_k^{l-1} + b_j^l
\end{equation}

\subsection{Step 2: Applying the Chain Rule}
We want to compute:
\begin{equation}
    \frac{\partial C}{\partial w_{jk}^l}
\end{equation}
Using the chain rule:
\begin{equation}
    \frac{\partial C}{\partial w_{jk}^l} = \frac{\partial C}{\partial z_j^l} \cdot \frac{\partial z_j^l}{\partial w_{jk}^l}
\end{equation}

From the backpropagation equations:
\begin{equation}
    \delta_j^l = \frac{\partial C}{\partial z_j^l}
\end{equation}
Thus, we rewrite the equation as:
\begin{equation}
    \frac{\partial C}{\partial w_{jk}^l} = \delta_j^l \cdot \frac{\partial z_j^l}{\partial w_{jk}^l}
\end{equation}

\subsection{Step 3: Compute $\frac{\partial z_j^l}{\partial w_{jk}^l}$}
From the equation of $z_j^l$:
\begin{equation}
    z_j^l = \sum_k w_{jk}^{l} a_k^{l-1} + b_j^l
\end{equation}
Taking the derivative with respect to $w_{jk}^l$:
\begin{equation}
    \frac{\partial z_j^l}{\partial w_{jk}^l} = a_k^{l-1}
\end{equation}

\subsection{Step 4: Substituting}
Substituting this result back:
\begin{equation}
    \frac{\partial C}{\partial w_{jk}^l} = \delta_j^l \cdot a_k^{l-1}
\end{equation}
which simplifies to:
\begin{equation}
    \frac{\partial C}{\partial w_{jk}^l} = a_k^{l-1} \delta_j^l
\end{equation}

This proves the formula asked of us for backpropagation.
